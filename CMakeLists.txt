cmake_minimum_required(VERSION 3.20)
project(edge_panel VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/third_party/lvgl/CMakeLists.txt")
  message(FATAL_ERROR "Initialize LVGL: git submodule update --init --recursive")
endif()
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/third_party/lvgl/include/lvgl/lv_version.h" LVGL_VERSION_HEADER)
if(NOT LVGL_VERSION_HEADER MATCHES "LVGL_VERSION_MAJOR 9" OR
   NOT LVGL_VERSION_HEADER MATCHES "LVGL_VERSION_MINOR 6" OR
   NOT LVGL_VERSION_HEADER MATCHES "LVGL_VERSION_PATCH 0")
  message(FATAL_ERROR "This project requires LVGL v9.6.0")
endif()
find_package(PkgConfig REQUIRED)
set(EDGE_PANEL_BACKEND "SDL" CACHE STRING "Display backend: SDL or DRM")
set_property(CACHE EDGE_PANEL_BACKEND PROPERTY STRINGS SDL DRM)
if(EDGE_PANEL_BACKEND STREQUAL "SDL")
    pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET sdl2)
    set(EDGE_PANEL_PLATFORM_SOURCE src/platform/sdl/sdl_platform.cpp)
    set(EDGE_PANEL_PLATFORM_LIBRARY PkgConfig::SDL2)
elseif(EDGE_PANEL_BACKEND STREQUAL "DRM")
    pkg_check_modules(DRM REQUIRED IMPORTED_TARGET libdrm)
    set(EDGE_PANEL_PLATFORM_SOURCE src/platform/linux/drm_platform.cpp src/platform/linux/evdev_touch.cpp)
    set(EDGE_PANEL_PLATFORM_LIBRARY PkgConfig::DRM)
else()
    message(FATAL_ERROR "EDGE_PANEL_BACKEND must be SDL or DRM")
endif()
find_package(Freetype REQUIRED)
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/config/lv_conf.h" CACHE PATH "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_FREETYPE ON CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
add_subdirectory(third_party/lvgl)
add_subdirectory(third_party/material-color-utilities)
set(EDGE_PANEL_THEME_SOURCES src/ui/theme/material_color.cpp src/ui/theme/color_scheme.cpp)
add_executable(edge-panel src/main.cpp ${EDGE_PANEL_PLATFORM_SOURCE} src/services/mock_provider.cpp src/ui/panel.cpp ${EDGE_PANEL_THEME_SOURCES})
target_include_directories(edge-panel PRIVATE src)
target_link_libraries(lvgl PUBLIC Freetype::Freetype)
target_link_libraries(edge-panel PRIVATE lvgl ${EDGE_PANEL_PLATFORM_LIBRARY} material-color-utilities)
target_compile_options(edge-panel PRIVATE -Wall -Wextra -Wpedantic)
if(EDGE_PANEL_BACKEND STREQUAL "DRM")
    target_compile_definitions(edge-panel PRIVATE EDGE_PANEL_BACKEND_DRM=1)
endif()
include(CTest)
# Cross-compiled executables cannot run on the build host without an emulator.
if(BUILD_TESTING AND NOT CMAKE_CROSSCOMPILING)
if(EDGE_PANEL_BACKEND STREQUAL "SDL")
add_test(NAME mock_interactions COMMAND edge-panel --self-test --seed-hue 0 --screenshot ${CMAKE_CURRENT_BINARY_DIR}/mock-panel.bmp)
set_tests_properties(mock_interactions PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy")
endif()
# 配色数学不依赖 LVGL，可脱离图形环境直接测试。
add_executable(theme-test tests/theme_test.cpp ${EDGE_PANEL_THEME_SOURCES})
target_include_directories(theme-test PRIVATE src)
target_link_libraries(theme-test PRIVATE material-color-utilities)
target_compile_options(theme-test PRIVATE -Wall -Wextra -Wpedantic)
add_test(NAME dynamic_theme COMMAND theme-test)
if(EDGE_PANEL_BACKEND STREQUAL "SDL")
add_test(NAME seed_cli_validation COMMAND ${CMAKE_COMMAND}
    -DPANEL_EXECUTABLE=$<TARGET_FILE:edge-panel>
    -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/cli_seed_test.cmake)
set_tests_properties(seed_cli_validation PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy")
endif()
add_executable(coordinates-test tests/coordinates_test.cpp)
target_include_directories(coordinates-test PRIVATE src)
target_compile_options(coordinates-test PRIVATE -Wall -Wextra -Wpedantic)
add_test(NAME display_touch_coordinates COMMAND coordinates-test)
if(EDGE_PANEL_BACKEND STREQUAL "DRM")
add_test(NAME drm_cli_failures COMMAND ${CMAKE_COMMAND}
    -DPANEL_EXECUTABLE=$<TARGET_FILE:edge-panel>
    -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/drm_cli_test.cmake)
endif()
endif()
